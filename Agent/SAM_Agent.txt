# ==========================================
# UTILITY
# ==========================================
function Write-SAMLog {
    param([string]$Message, [string]$Level = "INFO")
    $Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $Line = "$Timestamp [$Level] [Agent] $Message"
    try { Add-Content -Path $LogFile -Value $Line -Encoding UTF8 -ErrorAction SilentlyContinue } catch {}
    if (-not $Silent) { Write-Host $Line -ForegroundColor Cyan }
}

function Write-SAMChangeLog {
    # บันทึกเหตุการณ์เปลี่ยนแปลงลง Log แยก (เก็บไว้ถาวรกว่า 10 วัน = เก็บทุก event)
    param([string]$FieldName, [string]$OldValue, [string]$NewValue)
    $ChangeLogFile = "$WorkDir\SAM_ChangeLog.txt"
    $Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $Line = "$Timestamp [CHANGE] $FieldName | OLD: $OldValue | NEW: $NewValue"
    try { Add-Content -Path $ChangeLogFile -Value $Line -Encoding UTF8 -ErrorAction SilentlyContinue } catch {}
    Write-SAMLog "Change detected: [$FieldName] $OldValue → $NewValue" -Level "CHANGE"
}

function Get-FieldHash {
    param([string]$Value)
    $MD5 = New-Object Security.Cryptography.MD5CryptoServiceProvider
    return [BitConverter]::ToString(
        $MD5.ComputeHash([Text.Encoding]::UTF8.GetBytes($Value))
    ) -replace '-'
}
# ==========================================
# MONITOR DETECTION
# ==========================================
function Get-MonitorInfo {
    try {
        $Monitors = Get-CimInstance -Namespace root\wmi -ClassName WmiMonitorID -ErrorAction Stop
        if (-not $Monitors -or $Monitors.Count -eq 0) { return $null }

        $MonitorList = @()
        foreach ($M in $Monitors) {
            $Manufacturer = if ($M.ManufacturerName) {
                (($M.ManufacturerName | Where-Object { $_ -ne 0 -and $_ -ne $null } | ForEach-Object { [char]$_ }) -join '').Trim()
            } else { "" }

            $Model = if ($M.UserFriendlyName) {
                (($M.UserFriendlyName | Where-Object { $_ -ne 0 -and $_ -ne $null } | ForEach-Object { [char]$_ }) -join '').Trim()
            } else { "" }

            $Serial = if ($M.SerialNumberID) {
                (($M.SerialNumberID | Where-Object { $_ -ne 0 -and $_ -ne $null } | ForEach-Object { [char]$_ }) -join '').Trim()
            } else { "" }

            if ([string]::IsNullOrWhiteSpace($Manufacturer)) { $Manufacturer = "N/A" }
            if ([string]::IsNullOrWhiteSpace($Model))        { $Model        = "N/A" }
            if ([string]::IsNullOrWhiteSpace($Serial))       { $Serial       = "N/A" }

            $Year = if ($M.YearOfManufacture -and $M.YearOfManufacture -gt 1990) { $M.YearOfManufacture } else { $null }
            $Week = if ($M.WeekOfManufacture -and $M.WeekOfManufacture -gt 0 -and $M.WeekOfManufacture -le 53) { "{0:D2}" -f [int]$M.WeekOfManufacture } else { $null }
            $YearWeek = if ($Year -and $Week) { "$Year-$Week" } else { "N/A" }

            $MonitorList += "$Manufacturer|$Model|$Serial|$YearWeek"
        }

        if ($MonitorList.Count -eq 0) { return $null }
        return ($MonitorList -join ' ; ')
    } catch {
        Write-SAMLog "Monitor detection failed: $($_.Exception.Message)" -Level "WARN"
        return $null
    }
}

# ==========================================
# C2 HELPER: Update Status ใน sam_c2_commands
# ==========================================
function Update-C2Status {
    param(
        [string]$CommandId,
        [string]$Status,
        [string]$ResultMessage = ""
    )
    $body = @{ status = $Status }
    switch ($Status) {
        "acknowledged" { $body.acknowledged_at = (Get-Date).ToUniversalTime().ToString("o") }
        "done"         { $body.executed_at     = (Get-Date).ToUniversalTime().ToString("o") }
        "failed"       { $body.executed_at     = (Get-Date).ToUniversalTime().ToString("o") }
    }
    if ($ResultMessage) { $body.result_message = $ResultMessage }

    try {
        $hdr = @{
            "apikey"        = $SupabaseKey
            "Authorization" = "Bearer $SupabaseKey"
            "Content-Type"  = "application/json"
            "Prefer"        = "return=minimal"
        }
        Invoke-RestMethod `
            -Uri     "$SupabaseBase/sam_c2_commands?id=eq.$CommandId" `
            -Method  Patch `
            -Headers $hdr `
            -Body    ($body | ConvertTo-Json -Compress) `
            -TimeoutSec 10 | Out-Null
    } catch {
        Write-SAMLog "[C2] Status update failed [$CommandId]: $($_.Exception.Message)" -Level "WARN"
    }
}

# ==========================================
# C2 HELPER: ดึงข้อความ Notify จาก sam_notify_messages
# ==========================================
function Get-NotifyMessage {
    param([string]$NotifyId)  # "RESTART_NOTIFY" หรือ "SHUTDOWN_NOTIFY"

    $defaults = @{
        title           = "SAM C2: $NotifyId"
        message         = "เครื่องนี้จะดำเนินการใน 5 นาที"
        timeout_minutes = 5
    }

    try {
        $hdr = @{
            "apikey"        = $SupabaseKey
            "Authorization" = "Bearer $SupabaseKey"
        }
        $nmUrl = "$SupabaseBase/sam_notify_messages?id=eq.$NotifyId&select=title,message,timeout_minutes"
        $nmData = Invoke-RestMethod -Uri $nmUrl -Method Get -Headers $hdr -TimeoutSec 10

        if ($nmData -and $nmData.Count -gt 0) {
            $row = $nmData[0]
            if ($row.title)           { $defaults.title           = $row.title }
            if ($row.message)         { $defaults.message         = $row.message }
            if ($row.timeout_minutes) { $defaults.timeout_minutes = [int]$row.timeout_minutes }
            Write-SAMLog "[C2] Notify message loaded from Supabase: $NotifyId"
        } else {
            Write-SAMLog "[C2] No notify message found for $NotifyId, using defaults" -Level "WARN"
        }
    } catch {
        Write-SAMLog "[C2] Failed to fetch notify message: $($_.Exception.Message), using defaults" -Level "WARN"
    }

    return $defaults
}

# ==========================================
# C2 COMMAND PROCESSOR (จาก sam_c2_commands)
# ==========================================
function Invoke-C2Command {
    param([PSCustomObject]$Cmd)

    $id      = $Cmd.id
    $command = $Cmd.command
    $task    = $Cmd.task_filename
    $params  = if ($Cmd.task_parameters) {
                   try { $Cmd.task_parameters | ConvertFrom-Json } catch { $null }
               } else { $null }

    Write-SAMLog "[C2] Execute: [$command] id=$id"
    Update-C2Status -CommandId $id -Status "acknowledged"
    Update-C2Status -CommandId $id -Status "running"

    $result  = "OK"
    $success = $true

    try {
        switch ($command) {

            "CLEAR_TEMP" {
                $paths = @($env:TEMP, "C:\Windows\Temp")
                foreach ($p in $paths) {
                    if (Test-Path $p) {
                        Get-ChildItem $p -Recurse -Force -ErrorAction SilentlyContinue |
                            Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
                    }
                }
                $result = "CLEAR_TEMP completed"
            }

            "FORCE_SYNC" {
                Remove-Item $StateFile -Force -ErrorAction SilentlyContinue
                $result = "State cleared — FullSync next cycle"
            }

            "RESTART" {
                $result = "RESTART initiated"
                Update-C2Status -CommandId $id -Status "done" -ResultMessage $result
                Start-Sleep -Seconds 5
                Start-Process "shutdown.exe" -ArgumentList "/r /t 30 /c `"SAM Remote Restart`"" -WindowStyle Hidden
                return
            }

            "SHUTDOWN" {
                $result = "SHUTDOWN initiated"
                Update-C2Status -CommandId $id -Status "done" -ResultMessage $result
                Start-Sleep -Seconds 5
                Start-Process "shutdown.exe" -ArgumentList "/s /t 30 /c `"SAM Remote Shutdown`"" -WindowStyle Hidden
                return
            }

            # =============================================
            # RESTART_NOTIFY — แจ้งเตือนก่อน Restart
            # ดึงข้อความจาก Supabase sam_notify_messages
            # =============================================
            "RESTART_NOTIFY" {
                $notify = Get-NotifyMessage -NotifyId "RESTART_NOTIFY"
                $timeoutSec = $notify.timeout_minutes * 60

                Write-SAMLog "[C2] RESTART_NOTIFY: Title=$($notify.title) | Msg=$($notify.message) | Timeout=$($notify.timeout_minutes)min"

                # แจ้ง user ผ่าน msg command (ถ้ามี Active Session)
                $sessionCheck = query session 2>$null
                if ($sessionCheck -match "Active|^>") {
                    msg * "$($notify.title)`n$($notify.message)`nRestart ใน $($notify.timeout_minutes) นาที" /TIME:$timeoutSec 2>$null
                }

                # อัปเดต status ก่อน shutdown (เพราะหลัง restart อาจส่งไม่ทัน)
                $result = "OK: RESTART_NOTIFY - Notified, restart in $($notify.timeout_minutes) minutes"
                Update-C2Status -CommandId $id -Status "done" -ResultMessage $result
                Write-SAMLog "[C2] Result [RESTART_NOTIFY]: $result"

                # Schedule restart
                Start-Process "shutdown.exe" -ArgumentList "/r /t $timeoutSec /c `"$($notify.title) - $($notify.message)`"" `
                    -WindowStyle Hidden -NoNewWindow
                return
            }

            # =============================================
            # SHUTDOWN_NOTIFY — แจ้งเตือนก่อน Shutdown
            # ดึงข้อความจาก Supabase sam_notify_messages
            # =============================================
            "SHUTDOWN_NOTIFY" {
                $notify = Get-NotifyMessage -NotifyId "SHUTDOWN_NOTIFY"
                $timeoutSec = $notify.timeout_minutes * 60

                Write-SAMLog "[C2] SHUTDOWN_NOTIFY: Title=$($notify.title) | Msg=$($notify.message) | Timeout=$($notify.timeout_minutes)min"

                # แจ้ง user ผ่าน msg command
                $sessionCheck = query session 2>$null
                if ($sessionCheck -match "Active|^>") {
                    msg * "$($notify.title)`n$($notify.message)`nShutdown ใน $($notify.timeout_minutes) นาที" /TIME:$timeoutSec 2>$null
                }

                # อัปเดต status ก่อน shutdown
                $result = "OK: SHUTDOWN_NOTIFY - Notified, shutdown in $($notify.timeout_minutes) minutes"
                Update-C2Status -CommandId $id -Status "done" -ResultMessage $result
                Write-SAMLog "[C2] Result [SHUTDOWN_NOTIFY]: $result"

                # Schedule shutdown
                Start-Process "shutdown.exe" -ArgumentList "/s /t $timeoutSec /c `"$($notify.title) - $($notify.message)`"" `
                    -WindowStyle Hidden -NoNewWindow
                return
            }

            "RUN_TASK" {
                if (-not $task) { throw "task_filename is empty" }

                # ป้องกัน path traversal
                if ($task -notmatch '^[\w\-]+\.ps1$') { throw "Invalid task_filename: $task" }

                $taskUrl  = "$TaskBaseUrl$task"
                $taskPath = "$env:TEMP\$task"

                Invoke-WebRequest -Uri $taskUrl -OutFile $taskPath -UseBasicParsing -ErrorAction Stop
                if (-not (Test-Path $taskPath)) { throw "Download failed: $task" }

                # สร้าง argument string จาก task_parameters (ถ้ามี)
                $argStr = ""
                if ($params) {
                    $params.PSObject.Properties | ForEach-Object {
                        $argStr += " -$($_.Name) `"$($_.Value)`""
                    }
                }

                $PSA = "-WindowStyle Hidden -ExecutionPolicy Bypass -Command `"& { " +
                       "try { & '$taskPath'$argStr } " +
                       "finally { Start-Sleep 2; Remove-Item '$taskPath' -Force -ErrorAction SilentlyContinue } }`""
                Start-Process powershell.exe -ArgumentList $PSA -WindowStyle Hidden
                $result = "Task dispatched: $task$argStr"
            }

            default { throw "Unknown command: $command" }
        }
    } catch {
        $success = $false
        $result  = "ERROR: $($_.Exception.Message)"
        Write-SAMLog "[C2] Failed [$command]: $($_.Exception.Message)" -Level "WARN"
    }

    $finalStatus = if ($success) { "done" } else { "failed" }
    Update-C2Status -CommandId $id -Status $finalStatus -ResultMessage $result
    Write-SAMLog "[C2] Result [$command]: $result"
}

# ==========================================
# C2-ONLY MODE: poll จาก sam_c2_commands
# ==========================================
if ($C2Only) {
    Write-SAMLog "=== C2 Poll Mode ==="
    if (-not (Test-Path $ConfigFile)) {
        Write-SAMLog "No config, skip C2." -Level "WARN"; exit 0
    }

    # ดึง SerialNumber ของเครื่องนี้
    $BIOS = Get-CimInstance Win32_BIOS -ErrorAction SilentlyContinue
    $RawSN = if ($BIOS.SerialNumber) { $BIOS.SerialNumber.Trim() } else { "" }
    if ([string]::IsNullOrWhiteSpace($RawSN) -or $RawSN -match "(?i)To be filled|O\.E\.M\.|Default|None") {
        $UUID = (Get-CimInstance Win32_ComputerSystemProduct).UUID
        $SafeSerial = "HWID-" + ($UUID.Substring(0, 18))
    } else { $SafeSerial = $RawSN }

    # Poll pending commands
    try {
        $filter = "serial_number=eq.$([uri]::EscapeDataString($SafeSerial))" +
                  "&computer_name=eq.$([uri]::EscapeDataString($env:COMPUTERNAME))" +
                  "&status=eq.pending" +
                  "&order=batch_id.asc,seq.asc"

        $hdr = @{
            "apikey"        = $SupabaseKey
            "Authorization" = "Bearer $SupabaseKey"
            "User-Agent"    = "SAM-Agent/v$SAM_VERSION"
        }

        $commands = Invoke-RestMethod `
            -Uri     "$SupabaseBase/sam_c2_commands?$filter" `
            -Method  Get `
            -Headers $hdr `
            -TimeoutSec 15 `
            -ErrorAction Stop

        if ($commands -and $commands.Count -gt 0) {
            Write-SAMLog "[C2] Found $($commands.Count) pending command(s)"

            # Group by batch_id → execute ตาม seq
            $batches = $commands | Group-Object -Property batch_id
            foreach ($batch in $batches) {
                Write-SAMLog "[C2] Batch: $($batch.Name) ($($batch.Count) cmd)"
                $batch.Group | Sort-Object seq | ForEach-Object {
                    Invoke-C2Command -Cmd $_
                    Start-Sleep -Seconds 2
                }
            }
        } else {
            Write-SAMLog "[C2] No pending commands"
        }
    } catch {
        Write-SAMLog "[C2] Poll failed: $($_.Exception.Message)" -Level "WARN"
    }

    # Heartbeat
    try {
        $NowUTC = (Get-Date).ToUniversalTime()
        $NowTH  = $NowUTC.AddHours(7).ToString("yyyy-MM-dd HH:mm:ss")
        $pingUrl  = "$SupabaseBase/sam_devices" +
                    "?SerialNumber=eq.$([uri]::EscapeDataString($SafeSerial))"
        $pingBody = @{
            updated_at = $NowUTC.ToString("yyyy-MM-ddTHH:mm:ss.fffZ")
            LastSyncTH = $NowTH
        } | ConvertTo-Json -Compress
        $pingHdr = @{
            "apikey"        = $SupabaseKey
            "Authorization" = "Bearer $SupabaseKey"
            "Content-Type"  = "application/json"
            "Prefer"        = "return=minimal"
        }
        Invoke-RestMethod -Uri $pingUrl -Method Patch -Headers $pingHdr `
                          -Body $pingBody -TimeoutSec 10 | Out-Null
        Write-SAMLog "[C2] Heartbeat updated: $NowTH"
    } catch {
        Write-SAMLog "[C2] Heartbeat failed: $($_.Exception.Message)" -Level "WARN"
    }

    exit 0
}

        default {

            elseif ($Speed -ge 800)  { 'DDR3' }
            elseif ($Speed -ge 400)  { 'DDR2' }
            else                     { 'DDR'  }
        }

    # ============================================================
    # Slot Index Detection — Priority order
    # ============================================================
    $SlotIdx = -1
 
    # Priority 1: ChannelX-DIMMX (e.g. ChannelA-DIMM0, ChannelB-DIMM1)
    if ($Locator -match 'Channel([A-Da-d])[\-_\s]?DIMM[\-_\s]?(\d+)') {
        $chNum   = [int]([char]($matches[1].ToUpper()) - [char]'A')
        $dimmNum = [int]$matches[2]
        $SlotIdx = $chNum * 2 + $dimmNum
    }
    # Priority 2: DIMMX (e.g. DIMM0, DIMM1)
    elseif ($Locator -match 'DIMM[\-_\s]?(\d+)') {
        $SlotIdx = [int]$matches[1]
    }
    # Priority 3: SlotX (e.g. Slot 1, Slot#2)
    elseif ($Locator -match 'Slot[\s#_]?(\d+)') {
        $SlotIdx = [int]$matches[1] - 1
    }
    # Priority 4: BANKX จาก BankLabel (e.g. BANK 0, BANK 1)
    elseif ($Bank -match 'BANK[\s#_]?(\d+)') {
        $SlotIdx = [int]$matches[1]
    }
    # Priority 5: ตัวเลขท้าย Locator
    elseif ($Locator -match '(\d+)$') {
        $SlotIdx = [int]$matches[1]
    }
 
    # ถ้า SlotIdx เกินช่วง → หาช่องว่างแรก
    if ($SlotIdx -lt 0 -or $SlotIdx -ge $TotalSlots) {
        for ($j = 0; $j -lt $TotalSlots; $j++) {
            if ($null -eq $SlotInfos[$j]) { $SlotIdx = $j; break }
        }
    }
 
    # บันทึก — ป้องกัน overwrite
    if ($SlotIdx -ge 0 -and $SlotIdx -lt $TotalSlots) {
        if ($null -eq $SlotInfos[$SlotIdx]) {
            $SlotInfos[$SlotIdx] = @{ Loc = $Locator; Info = $InfoStr }
        } else {
            # Slot เต็ม → หาช่องถัดไป
            for ($j = 0; $j -lt $TotalSlots; $j++) {
                if ($null -eq $SlotInfos[$j]) {
                    $SlotInfos[$j] = @{ Loc = $Locator; Info = $InfoStr }
                    break
                }
            }
        }
    }
}

function Get-AVVersion {
    param([string]$AVName)
    try {
        if ($AVName -match "Defender|Microsoft") {
            $mpStatus = Get-MpComputerStatus -ErrorAction SilentlyContinue
            if ($mpStatus -and $mpStatus.AMProductVersion) { return $mpStatus.AMProductVersion }
        }
        if ($AVName -match "Trend Micro|Apex One|OfficeScan|PC-cillin") {
            $tmExeList = @(
                "$env:ProgramFiles\Trend Micro\Security Agent\Ntrtscan.exe",
                "${env:ProgramFiles(x86)}\Trend Micro\Security Agent\Ntrtscan.exe",
                "$env:ProgramFiles\Trend Micro\Apex One Security Agent\Ntrtscan.exe",
                "${env:ProgramFiles(x86)}\Trend Micro\OfficeScan Client\Ntrtscan.exe"
            )
            $tmExe = $tmExeList | Where-Object { Test-Path $_ } | Select-Object -First 1
            if ($tmExe) {
                $rawVer = (Get-Item $tmExe -ErrorAction SilentlyContinue).VersionInfo.FileVersion
                if ($rawVer) { return ($rawVer -replace '^(\d+\.\d+)\.0\.(\d+)$','$1.$2') }
            }
            foreach ($p in @("HKLM:\SOFTWARE\TrendMicro\PC-cillinNTCorp\CurrentVersion","HKLM:\SOFTWARE\WOW6432Node\TrendMicro\PC-cillinNTCorp\CurrentVersion")) {
                if (Test-Path $p) {
                    $reg = Get-ItemProperty $p -ErrorAction SilentlyContinue
                    $ver = if ($reg.ProgramVer) { $reg.ProgramVer } elseif ($reg.AgentVersion) { $reg.AgentVersion } else { $null }
                    if ($ver) { return $ver }
                }
            }
        }
        $searchName = $AVName -replace " Antivirus| Security| Endpoint| Agent",""
        $swPaths = @("HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*",
                     "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*")
        $found = Get-ItemProperty $swPaths -ErrorAction SilentlyContinue |
            Where-Object { $_.DisplayName -and $_.DisplayName -match [regex]::Escape($searchName) } |
            Select-Object -First 1
        if ($found -and $found.DisplayVersion) { return $found.DisplayVersion }
    } catch {}
    return "Unknown"
}

                    if ($PP -and $PP.PrinterHostAddress) { $PIP = "IP:$($PP.PrinterHostAddress)" }
                } catch {}
                if ($PIP) { $PIP } else { $PPort }
            }
        }
        $PrinterList += "$PName [$PortDisp]"
    }
} catch {
    try {
        Get-CimInstance Win32_Printer -ErrorAction SilentlyContinue | ForEach-Object {
            $PrinterList += "$($_.Name) [$($_.PortName)]"
        }
    } catch {}
}

$OposPaths = @('HKLM:\SOFTWARE\OLEforRetail\ServiceOPOS','HKLM:\SOFTWARE\WOW6432Node\OLEforRetail\ServiceOPOS')
$OposSeen = @{}
foreach ($oBase in $OposPaths) {
    if (-not (Test-Path $oBase)) { continue }
    try {
        Get-ChildItem $oBase -ErrorAction SilentlyContinue | ForEach-Object {
            $oType = $_.PSChildName
            Get-ChildItem $_.PSPath -ErrorAction SilentlyContinue | ForEach-Object {
                $oName = $_.PSChildName
                $oKey  = "$oType`:$oName"
                if (-not $OposSeen.ContainsKey($oKey)) {
                    $OposSeen[$oKey] = $true
                    $oLabel = switch ($oType) {
                        'POSPrinter'  { 'OPOS' }
                        'CashDrawer'  { 'OPOS:Drawer' }
                        'LineDisplay' { 'OPOS:Display' }
                        'Scanner'     { 'OPOS:Scanner' }
                        'MSR'         { 'OPOS:MSR' }
                        'Keylock'     { 'OPOS:Keylock' }
                        default       { "OPOS:$oType" }
                    }
                    $PrinterList += "$oName [$oLabel]"
                }
            }
        }
    } catch {}
}
$CloudPrinters = $PrinterList -join " | "

$DiskList = @()
if (Test-Path $SmartCtlPath) {
    try {
        $ScanJ = & $SmartCtlPath --scan -j 2>$null | ConvertFrom-Json
        $Seen  = @()
        foreach ($Dev in $ScanJ.devices) {
            $DJ = & $SmartCtlPath -a $Dev.name -j 2>$null | ConvertFrom-Json
            if (-not $DJ -or -not $DJ.model_name) { continue }
            if ($Seen -contains $DJ.serial_number) { continue }
            $Seen += $DJ.serial_number
            $Type    = if($DJ.device.protocol -match "NVMe" -or $DJ.rotation_rate -eq 0){"SSD"}else{"HDD"}
            $Status  = if($DJ.smart_status.passed){"Healthy"}else{"Critical"}
            $Temp    = if($DJ.temperature.current){"$($DJ.temperature.current)°C"}else{"N/A"}
            $Hours   = if($DJ.power_on_time.hours){"$($DJ.power_on_time.hours) hrs"}else{"N/A"}
            $Health  = "N/A"
            if ($Type -eq "SSD") {
                $HP = $null
                if ($null -ne $DJ.nvme_smart_health_information_log.percentage_used) { $HP = 100 - $DJ.nvme_smart_health_information_log.percentage_used }
                elseif ($DJ.ata_smart_attributes.table) {
                    $A = $DJ.ata_smart_attributes.table | Where-Object { $_.id -in @(177,202,230,231,233) } | Select-Object -First 1
                    if ($A) { $HP = if($DJ.model_name -match "WDC|WD " -and $A.id -eq 230){100-$A.value}else{$A.value} }
                }
                if ($null -ne $HP) { $Health = if($HP -ge 30){"✅ Good($HP%)"}elseif($HP -gt 10){"⚠️ Warn($HP%)"}else{"❌ Bad($HP%)"} }
            }
            $DiskList += "Model:$($DJ.model_name) | Type:$Type | Status:$Status | Temp:$Temp | Health:$Health | Hrs:$Hours"
        }
    } catch { Write-SAMLog "S.M.A.R.T scan error: $($_.Exception.Message)" -Level "WARN" }
}
if ($DiskList.Count -eq 0) {
    try {
        Get-PhysicalDisk -ErrorAction Stop | ForEach-Object {
            $R = Get-StorageReliabilityCounter -PhysicalDisk $_ -ErrorAction SilentlyContinue
            $DiskList += "Model:$($_.Model) | Type:$($_.MediaType) | Status:$($_.HealthStatus) | Wear:$(if($R){$R.Wear}else{'N/A'})%"
        }
    } catch { $DiskList += "Unavailable" }
}
$CloudDisk = $DiskList -join "`n"

$DriveSpaceList = @()
try {
    Get-CimInstance Win32_LogicalDisk -Filter "DriveType=3" | ForEach-Object {
        $TB = [math]::Round($_.Size/1GB,1)
        $FB = [math]::Round($_.FreeSpace/1GB,1)
        $FP = if($TB -gt 0){[math]::Round(($FB/$TB)*100,1)}else{0}
        $Ico = if($FP -le 10){"❌"}elseif($FP -le 20){"⚠️"}else{"✅"}
        $DriveSpaceList += "$Ico [$($_.DeviceID)] $FP% ($FB/$TB GB)"
    }
} catch {}
$CloudDriveSpace = $DriveSpaceList -join " | "

$CloudNet = "N/A"; $MAC = "N/A"
try {
    $Net = Get-WmiObject Win32_NetworkAdapterConfiguration | Where-Object { $_.IPAddress -and $_.DefaultIPGateway } | Select-Object -First 1
    if ($Net) {
        $IP = $Net.IPAddress[0]; $MAC = $Net.MACAddress
        $DNS = if($Net.DNSServerSearchOrder){$Net.DNSServerSearchOrder -join ", "}else{"N/A"}
        $CloudNet = "$IP (SN:$($Net.IPSubnet[0]) | GW:$($Net.DefaultIPGateway[0]) | DNS:$DNS)"
    }
} catch {}

$Stability    = (Get-CimInstance Win32_ReliabilityStabilityMetrics -ErrorAction SilentlyContinue | Sort-Object timegenerated -Descending | Select-Object -First 1).SystemStabilityIndex
$HealthScore  = if($null -ne $Stability){"$([math]::Round($Stability,1))/10"}else{"N/A"}
$BootTime     = (Get-CimInstance Win32_OperatingSystem).LastBootUpTime
$UptimeDays   = ((Get-Date) - $BootTime).Days
$UptimeStr    = if($UptimeDays -ge 14){"$UptimeDays วัน (⚠️ ควร Restart)"}else{"$UptimeDays วัน"}
$P1 = Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending"
$P2 = Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
$P3 = $null -ne (Get-ItemProperty "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager" -Name PendingFileRenameOperations -ErrorAction SilentlyContinue)
$PendingRb    = if($P1 -or $P2 -or $P3){"⚠️ Requires Restart"}else{"✅ Clear"}

$NowUTC  = (Get-Date).ToUniversalTime()
$NowTH   = $NowUTC.AddHours(7)

$CurrentData = [ordered]@{

# ==========================================
# SMART DELTA SYNC ENGINE
# ==========================================
Write-SAMLog "Analyzing changes (Delta Engine)..."

$LastState     = $null
$LastFieldHash = @{}
$LastFullSync  = $null

if (Test-Path $StateFile) {
    try {
        $LastState    = Get-Content $StateFile -Raw | ConvertFrom-Json
        $LastFullSync = if ($LastState.LastFullSync) { [datetime]$LastState.LastFullSync } else { $null }
        if ($LastState.FieldHashes) {
            $LastState.FieldHashes.PSObject.Properties | ForEach-Object {
                $LastFieldHash[$_.Name] = $_.Value
            }
        }
    } catch { Write-SAMLog "State file corrupt, will force FullSync." -Level "WARN" }
}

$DaysSinceFullSync = if ($LastFullSync) { ((Get-Date) - $LastFullSync).TotalDays } else { 999 }
$NeedFullSync      = $ForceFullSync -or ($DaysSinceFullSync -ge $FULL_SYNC_INTERVAL_DAYS) -or ($null -eq $LastState)

$CurrentFieldHash = @{}
$ChangedFields    = @()
$DeltaPayload     = [ordered]@{
    SerialNumber = $SafeSerial
    ComputerName = $env:COMPUTERNAME
    updated_at   = $CurrentData.updated_at
    LastSyncTH   = $CurrentData.LastSyncTH
    AgentVersion = $CurrentData.AgentVersion
}

foreach ($Key in $CurrentData.Keys) {
    $Val  = if ($null -ne $CurrentData[$Key]) { $CurrentData[$Key].ToString() } else { "" }
    $Hash = Get-FieldHash $Val
    $CurrentFieldHash[$Key] = $Hash

    if (-not $NeedFullSync) {
        $OldHash = $LastFieldHash[$Key]
        if ($OldHash -ne $Hash) {
            $ChangedFields += $Key
            $DeltaPayload[$Key] = $CurrentData[$Key]
            $OldVal = if ($LastState -and $LastState.FieldHashes -and $LastState.FieldData -and $LastState.FieldData.$Key) { $LastState.FieldData.$Key } else { "(unknown)" }
            Write-SAMChangeLog -FieldName $Key -OldValue $OldVal -NewValue $Val
        }
    }
}

if ($NeedFullSync) {
    $SyncType    = "FullSync"
    $PayloadData = $CurrentData
    $PayloadData["Type"] = "FullSync"
    Write-SAMLog "Sync type: FullSync (DaysSinceFull=$([math]::Round($DaysSinceFullSync,1)), Forced=$ForceFullSync)"
} elseif ($ChangedFields.Count -gt 0) {
    $SyncType    = "DeltaSync"
    $DeltaPayload["Type"] = "DeltaSync"
    $DeltaPayload["ChangedFields"] = ($ChangedFields -join ",")
    $PayloadData = $DeltaPayload
    Write-SAMLog "Sync type: DeltaSync — $($ChangedFields.Count) field(s) changed: $($ChangedFields -join ', ')"
} else {
    $SyncType = "Heartbeat"
    $DeltaPayload["Type"] = "Heartbeat"
    $PayloadData = $DeltaPayload
    Write-SAMLog "Sync type: Heartbeat (no changes detected)"
}

$Headers = @{
    "apikey"        = $SupabaseKey
    "Authorization" = "Bearer $SupabaseKey"
    "Content-Type"  = "application/json; charset=utf-8"
    "Prefer"        = "resolution=merge-duplicates, return=representation"
    "User-Agent"    = "SAM-Agent/v$SAM_VERSION"
}

$SupabaseFields = @(
    "BU","BranchID","DeviceType","Province","ComputerName","ComputerDesc","Manufacturer","Model",
    "SerialNumber","MonitorSN","OSVersion","WinUpdate","BootPerf","CPU","RAM","IPAddress","MACAddress",
    "Antivirus1","Antivirus2","Firewall","DiskHealth","DriveSpace","BitLocker","StabilityScore",
    "Uptime","PendingReboot","TeamViewerID","AnyDeskID","ActiveUsers","AllPrinters","AllSoftware",
    "AgentVersion","Type","updated_at","LastSyncTH"
)
$FilteredPayload = [ordered]@{}
foreach ($k in $PayloadData.Keys) {
    if ($SupabaseFields -contains $k) { $FilteredPayload[$k] = $PayloadData[$k] }
    else { Write-SAMLog "Skipped field not in schema: $k" -Level "WARN" }
}

function Remove-NullBytes {
    param([object]$Obj)
    if ($Obj -is [System.Collections.IDictionary]) {
        $clean = [ordered]@{}
        foreach ($k in $Obj.Keys) { $clean[$k] = Remove-NullBytes $Obj[$k] }
        return $clean
    } elseif ($Obj -is [string]) {
        return $Obj -replace '\u0000','' -replace [char]0,''
    } else { return $Obj }
}

$CleanPayload = Remove-NullBytes $FilteredPayload
$JsonPayload  = ($CleanPayload | ConvertTo-Json -Depth 5 -Compress) -replace '\\u0000','' -replace '\x00',''
$Utf8Bytes    = [System.Text.Encoding]::UTF8.GetBytes($JsonPayload)
$MaxRetries   = 6; $Attempt = 0; $IsSuccess = $false

while ($Attempt -lt $MaxRetries -and -not $IsSuccess) {
    $Attempt++
    try {
        $Timer = [System.Diagnostics.Stopwatch]::StartNew()
        $Response = Invoke-RestMethod -Uri $SupabaseUrl -Method Post -Headers $Headers -Body $Utf8Bytes -TimeoutSec 30 -ErrorAction Stop
        $Timer.Stop()
        $IsSuccess = $true

        $ElapsedS = [math]::Round($Timer.Elapsed.TotalSeconds, 2)
        $SizeKB   = [math]::Round($Utf8Bytes.Length / 1KB, 2)
        Write-SAMLog "Supabase sync OK. Type:$SyncType | Attempt:$Attempt | $SizeKB KB | $ElapsedS s"

        $NewState = @{
            Hash         = (Get-FieldHash $JsonPayload)
            LastFullSync = if ($SyncType -eq "FullSync") { (Get-Date).ToString("o") } else { if ($LastFullSync) { $LastFullSync.ToString("o") } else { "" } }
            LastWriteTime = (Get-Date).ToString("o")
            FieldHashes  = $CurrentFieldHash
            FieldData    = @{
                OSVersion = $CurrentData.OSVersion; RAM = $CurrentData.RAM; IPAddress = $CurrentData.IPAddress
                DriveSpace = $CurrentData.DriveSpace; DiskHealth = $CurrentData.DiskHealth
                Antivirus1 = $CurrentData.Antivirus1; WinUpdate = $CurrentData.WinUpdate
                DeviceType = $CurrentData.DeviceType
            }
        }
        try { $NewState | ConvertTo-Json -Depth 5 | Out-File $StateFile -Encoding UTF8 } catch {}

    } catch {
        if ($null -ne $Timer) { $Timer.Stop() }
        $ErrMsg = if ($_.ErrorDetails) { $_.ErrorDetails.Message } else { $_.Exception.Message }
        if ($_.Exception.Response) {
            try {
                $reader  = New-Object System.IO.StreamReader($_.Exception.Response.GetResponseStream())
                $body    = $reader.ReadToEnd()
                $ErrMsg += " | Supabase: $body"
            } catch {}
        }
        $RetryWait = Get-Random -Minimum 60 -Maximum 300
        Write-SAMLog "Sync failed (Attempt $Attempt/$MaxRetries): $ErrMsg. Retry in $RetryWait s." -Level "WARN"
        if ($Attempt -lt $MaxRetries) { Start-Sleep -Seconds $RetryWait }
    }
}

if (-not $IsSuccess) {
    Write-SAMLog "All $MaxRetries sync attempts failed." -Level "ERROR"
}

function Build-HTMLReport {
    $CleanBranch = $BranchID -replace '[\\/:*?"<>|]', '-'
    $ReportPath  = "$WorkDir\$($BU)_$($CleanBranch)_$($env:COMPUTERNAME)_$(Get-Date -Format 'yyyy-MM-dd').html"

    Get-ChildItem $WorkDir -Filter "$($BU)_$($CleanBranch)_$($env:COMPUTERNAME)*.html" |
        Where-Object { $_.FullName -ne $ReportPath } | Remove-Item -Force -ErrorAction SilentlyContinue

    $CSS = "<style>body{font-family:'Leelawadee UI',Tahoma,sans-serif;margin:20px}table{border-collapse:collapse;width:95%;margin:auto auto 25px;box-shadow:0 2px 3px rgba(0,0,0,.1)}th,td{border:1px solid #ddd;text-align:left;padding:8px}th{background-color:#f2f2f2}tr:nth-child(even){background-color:#f9f9f9}h1,h2{text-align:center;color:#333}h1{border-bottom:2px solid #eee;padding-bottom:10px}.delta{color:#0056b3;font-weight:bold}</style>"
    $H  = "<!DOCTYPE html><html><head><meta charset='UTF-8'>$CSS</head><body>"
    $H += "<h1>System Audit — $env:COMPUTERNAME<br><span style='font-size:0.6em;color:#666;'>BU:$BU | Branch:$BranchID | Prov:$Province | Agent:v$SAM_VERSION | Sync:$SyncType</span></h1>"

    $H += "<h2>System Information</h2><table>"

    $DisplayNameMap = @{
        "MonitorSN" = "MonitorConnect"
    }

    foreach ($K in $CurrentData.Keys | Where-Object { $_ -notin @("updated_at","LastSyncTH","Type","ChangedFields","AllSoftware") }) {
        $MarkClass = if ($ChangedFields -contains $K) { " class='delta'" } else { "" }
        $DisplayKey = if ($DisplayNameMap.ContainsKey($K)) { $DisplayNameMap[$K] } else { $K }
        $DisplayVal = if ($null -eq $CurrentData[$K] -or [string]::IsNullOrWhiteSpace($CurrentData[$K].ToString())) {
            "N/A"
        } else {
            $CurrentData[$K]
        }
        $H += "<tr><td$MarkClass>$DisplayKey</td><td$MarkClass>$DisplayVal</td></tr>"
    }
    $H += "</table>"

    if ($AllSW.Count -gt 0) {
        $H += "<h2>Installed Applications</h2><table><tr><th>Name</th><th>Version</th><th>Install Date</th></tr>"
        foreach ($App in $AllSW) {
            $H += "<tr><td>$($App.DisplayName)</td><td>$($App.DisplayVersion)</td><td>$($App.InstallDate)</td></tr>"
        }
        $H += "</table>"
    }

    $H += "</body></html>"
    try { [System.IO.File]::WriteAllText($ReportPath, $H, [System.Text.Encoding]::UTF8) } catch {}
    if (-not $Silent) { try { Invoke-Item $ReportPath } catch {} }
    Write-SAMLog "HTML report saved: $ReportPath"
}
