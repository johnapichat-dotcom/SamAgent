# ==========================================
# UTILITY: Write-SAMLog
# ==========================================
function Write-SAMLog {
    param([string]$Message, [string]$Level = "INFO")
    $Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $Line = "$Timestamp [$Level] [Scheduler] $Message"
    
    if (-not (Test-Path $LogDir)) {
        New-Item -ItemType Directory -Path $LogDir -Force | Out-Null
    }
    
    try {
        Add-Content -Path $LogFile -Value $Line -Encoding UTF8 -ErrorAction SilentlyContinue
    } catch {}
    
    if (-not $env:SAM_SILENT) { Write-Host $Line -ForegroundColor Cyan }
}

# ==========================================
# FOLDER BOOTSTRAP — PS 5.0 Compatible
# ==========================================
function Invoke-FolderBootstrap {
    Write-SAMLog "Bootstrapping working directory: $WorkDir"

    # === STEP 1: Create directories ===
    if (-not (Test-Path $WorkDir)) {
        try {
            New-Item -ItemType Directory -Force -Path $WorkDir | Out-Null
            Write-SAMLog "Created WorkDir: $WorkDir"
        } catch {
            Write-SAMLog "Failed to create $WorkDir : $_" -Level "ERROR"
            exit 1
        }
    }

    if (-not (Test-Path $LogDir)) {
        try {
            New-Item -ItemType Directory -Force -Path $LogDir | Out-Null
            Write-SAMLog "Created LogDir: $LogDir"
        } catch {
            Write-SAMLog "Failed to create $LogDir : $_" -Level "ERROR"
            exit 1
        }
    }

    # === STEP 2: Permission fixes (ASYNC — fire and forget) ===
    Write-SAMLog "Queueing permission/attribute fixes (async)..."
    
    # Script block for permission fixes
    $PermissionScript = {
        param($WorkDir)
        $CmdsToRun = @(
            @{ Exe = "takeown.exe"; Args = @("/f", $WorkDir, "/r", "/d", "y") },
            @{ Exe = "icacls.exe"; Args = @($WorkDir, "/reset", "/T", "/C", "/Q") },
            @{ Exe = "attrib.exe"; Args = @("-h", "-s", "-r", $WorkDir, "/s", "/d") }
        )
        
        foreach ($Cmd in $CmdsToRun) {
            try {
                $Proc = Start-Process -FilePath $Cmd.Exe -ArgumentList $Cmd.Args `
                    -NoNewWindow -PassThru -ErrorAction SilentlyContinue
                if ($Proc) {
                    # Wait max 3 seconds per command (PS 5.0 compatible)
                    $Stopwatch = [System.Diagnostics.Stopwatch]::StartNew()
                    while (-not $Proc.HasExited -and $Stopwatch.ElapsedMilliseconds -lt 3000) {
                        Start-Sleep -Milliseconds 100
                    }
                    if (-not $Proc.HasExited) { 
                        $Proc.Kill() 
                    }
                }
            }
            catch { }
        }
    }
    
    # Run async — fire and forget (does NOT block)
    try {
        Start-Job -ScriptBlock $PermissionScript -ArgumentList $WorkDir -ErrorAction SilentlyContinue | Out-Null
        Write-SAMLog "Permission fixes queued (background job)."
    }
    catch {
        # Fallback: if Start-Job unavailable, skip (non-critical)
        Write-SAMLog "Background job unavailable (non-critical): $($_.Exception.Message)" -Level "WARN"
    }

    # === STEP 3: Set folder attributes ===
    try {
        $Item = Get-Item $WorkDir -Force -ErrorAction SilentlyContinue
        if ($Item) {
            $Item.Attributes = 'Hidden, System'
            Write-SAMLog "Folder attributes set (hidden)."
        }
    } catch {}

    Write-SAMLog "Folder bootstrap complete."
}

# ==========================================
# HELPER: Register-SAMTask
# ==========================================
function Register-SAMTask {
    param(
        [string]$TaskName,
        [string]$ScriptFile,
        [string]$ExtraArgs = "",
        [object[]]$Triggers,
        [string]$Description = ""
    )

    $RunUser = "NT AUTHORITY\SYSTEM"
    $PSArgs  = "-WindowStyle Hidden -ExecutionPolicy Bypass -File `"$ScriptFile`""
    if ($ExtraArgs) { $PSArgs += " $ExtraArgs" }

    try {
        $Action   = New-ScheduledTaskAction -Execute "powershell.exe" -Argument $PSArgs
        $Settings = New-ScheduledTaskSettingsSet `
            -RunOnlyIfNetworkAvailable `
            -StartWhenAvailable `
            -AllowStartIfOnBatteries `
            -DontStopIfGoingOnBatteries `
            -ExecutionTimeLimit (New-TimeSpan -Minutes 30)

        # Remove existing task
        $ExistingTask = Get-ScheduledTask -TaskName $TaskName -ErrorAction SilentlyContinue
        if ($ExistingTask) {
            Unregister-ScheduledTask -TaskName $TaskName -Confirm:$false -ErrorAction SilentlyContinue | Out-Null
        }

        # Register task
        Register-ScheduledTask `
            -TaskName    $TaskName `
            -Action      $Action `
            -Trigger     $Triggers `
            -Settings    $Settings `
            -User        $RunUser `
            -RunLevel    Highest `
            -Description $Description `
            -Force | Out-Null

        Write-SAMLog "✓ Task registered: $TaskName"
    }
    catch {
        Write-SAMLog "Failed to register task $TaskName : $_" -Level "ERROR"
    }
}

# ==========================================
# INSTALL: ลงทะเบียน Scheduled Tasks
# ==========================================
function Install-SAMTasks {
    Write-SAMLog "=== Installing SAM Scheduled Tasks ==="

    # Remove legacy tasks
    foreach ($OldTask in @("JARVIS_Smart_Sync","SAM_Smart_Sync","SAM_Agent_DailySync","SAM_C2_Poll","SAM_FullSync_Daily")) {
        $Old = Get-ScheduledTask -TaskName $OldTask -ErrorAction SilentlyContinue
        if ($Old) {
            Unregister-ScheduledTask -TaskName $OldTask -Confirm:$false -ErrorAction SilentlyContinue | Out-Null
        }
    }

    # Generate random jitter for each machine
    $J_Agent  = Get-Random -Min 0 -Max 239
    $J_C2     = Get-Random -Min 0 -Max 39
    $J_Full   = Get-Random -Min 0 -Max 359
    $J_Update = Get-Random -Min 0 -Max 239
    $J_Log    = Get-Random -Min 0 -Max 59

    $BaseAgent  = [datetime]::Today.AddHours(6).AddMinutes($J_Agent)
    $BaseC2     = [datetime]::Today.AddHours(6).AddMinutes($J_C2)
    $BaseC2Ngt  = [datetime]::Today.AddHours(17).AddMinutes($J_C2)
    $BaseFull   = [datetime]::Today.AddHours(2).AddMinutes($J_Full)
    $BaseUpdate = [datetime]::Today.AddHours(9).AddMinutes($J_Update)
    $BaseLog    = [datetime]::Today.AddHours(10).AddMinutes($J_Log)

    Write-SAMLog "Jitter: Agent=$($BaseAgent.ToString('HH:mm')) C2Day=$($BaseC2.ToString('HH:mm')) C2Night=$($BaseC2Ngt.ToString('HH:mm')) Full=$($BaseFull.ToString('HH:mm')) Update=$($BaseUpdate.ToString('HH:mm')) Log=$($BaseLog.ToString('HH:mm'))"

    # TASK 1: Delta Sync + Startup
    $T1_Startup = New-ScheduledTaskTrigger -AtStartup
    $T1_Repeat  = New-ScheduledTaskTrigger -Once -At $BaseAgent `
                      -RepetitionInterval (New-TimeSpan -Hours 4) `
                      -RepetitionDuration (New-TimeSpan -Days 9999)
    Register-SAMTask `
        -TaskName    $TASK_AGENT_SYNC `
        -ScriptFile  $AgentScript `
        -ExtraArgs   "-Silent -DeltaOnly" `
        -Triggers    @($T1_Startup, $T1_Repeat) `
        -Description "SAM: Delta Sync every 4hr, start $($BaseAgent.ToString('HH:mm'))"

    # TASK 2: C2 Day
    $T2_Day = New-ScheduledTaskTrigger -Once -At $BaseC2 `
                  -RepetitionInterval (New-TimeSpan -Minutes 40) `
                  -RepetitionDuration (New-TimeSpan -Hours 11)
    Register-SAMTask `
        -TaskName    $TASK_C2_DAY `
        -ScriptFile  $AgentScript `
        -ExtraArgs   "-Silent -C2Only" `
        -Triggers    @($T2_Day) `
        -Description "SAM: C2 Poll day 40min, start $($BaseC2.ToString('HH:mm'))"

    # TASK 3: C2 Night
    $T3_Night = New-ScheduledTaskTrigger -Once -At $BaseC2Ngt `
                    -RepetitionInterval (New-TimeSpan -Minutes 80) `
                    -RepetitionDuration (New-TimeSpan -Hours 13)
    Register-SAMTask `
        -TaskName    $TASK_C2_NIGHT `
        -ScriptFile  $AgentScript `
        -ExtraArgs   "-Silent -C2Only" `
        -Triggers    @($T3_Night) `
        -Description "SAM: C2 Poll night 80min, start $($BaseC2Ngt.ToString('HH:mm'))"

    # TASK 4: Full Sync Weekly
    $T4 = New-ScheduledTaskTrigger -Weekly -DaysOfWeek Monday -At $BaseFull
    Register-SAMTask `
        -TaskName    $TASK_FULL_SYNC `
        -ScriptFile  $AgentScript `
        -ExtraArgs   "-Silent -ForceFullSync" `
        -Triggers    @($T4) `
        -Description "SAM: Weekly FullSync Monday $($BaseFull.ToString('HH:mm'))"

    # TASK 5: Update Check
    $T5 = New-ScheduledTaskTrigger -Daily -At $BaseUpdate
    Register-SAMTask `
        -TaskName    $TASK_UPDATE_CHECK `
        -ScriptFile  $UpdateScript `
        -ExtraArgs   "-Silent" `
        -Triggers    @($T5) `
        -Description "SAM: Update check daily $($BaseUpdate.ToString('HH:mm'))"

    # TASK 6: Log Rotation
    $T6 = New-ScheduledTaskTrigger -Daily -At $BaseLog
    Register-SAMTask `
        -TaskName    $TASK_LOG_ROTATE `
        -ScriptFile  $SchedulerScript `
        -ExtraArgs   "-LogRotateOnly" `
        -Triggers    @($T6) `
        -Description "SAM: Log rotation daily $($BaseLog.ToString('HH:mm'))"

    New-SAMHelperFiles
    Write-SAMLog "=== All SAM Tasks installed successfully ==="
}

# ==========================================
# LOG ROTATION
# ==========================================
function Invoke-LogRotate-File {
    param([string]$FilePath, [int]$RetainDays, [string]$Label)
    if (-not (Test-Path $FilePath)) { return }
    try {
        $Cutoff   = (Get-Date).AddDays(-$RetainDays)
        $AllLines = @(Get-Content $FilePath -Encoding UTF8 -ErrorAction SilentlyContinue)
        $Kept     = @($AllLines | Where-Object {
            if ($_ -match "^(\d{4}-\d{2}-\d{2})") {
                try { [datetime]::ParseExact($Matches[1],"yyyy-MM-dd",$null) -ge $Cutoff }
                catch { $true }
            } else { $true }
        })
        [System.IO.File]::WriteAllLines($FilePath, $Kept, [System.Text.Encoding]::UTF8)
        Write-SAMLog "$Label rotation: kept $($Kept.Count)/$($AllLines.Count) lines"
    } catch {
        Write-SAMLog "$Label rotation error: $($_.Exception.Message)" -Level "WARN"
    }
}

function Invoke-LogRotation {
    Invoke-LogRotate-File -FilePath "$LogDir\SAM_Scheduler.log" -RetainDays 7 -Label "SAM_Scheduler"
    Invoke-LogRotate-File -FilePath "$WorkDir\SAM_ChangeLog.txt" -RetainDays 15 -Label "SAM_ChangeLog"
    Invoke-LogRotate-File -FilePath "$WorkDir\SAM_Log_Offline.txt" -RetainDays 7 -Label "SAM_Log_Offline"

    try {
        $OldHTML = @(Get-ChildItem -Path $WorkDir -Filter "*.html" -ErrorAction SilentlyContinue | 
            Where-Object { $_.LastWriteTime -lt (Get-Date).AddDays(-7) })
        if ($OldHTML.Count -gt 0) {
            $OldHTML | Remove-Item -Force -ErrorAction SilentlyContinue
            Write-SAMLog "Removed $($OldHTML.Count) old HTML report(s)."
        }
    } catch {}

    try {
        $OfflineRptDir = "$WorkDir\OfflineReports"
        if (Test-Path $OfflineRptDir) {
            $OldJSON = @(Get-ChildItem -Path $OfflineRptDir -Filter "*.json" -ErrorAction SilentlyContinue |
                Where-Object { $_.LastWriteTime -lt (Get-Date).AddDays(-15) })
            if ($OldJSON.Count -gt 0) {
                $OldJSON | Remove-Item -Force -ErrorAction SilentlyContinue
                Write-SAMLog "Removed $($OldJSON.Count) old Offline JSON(s)."
            }
        }
    } catch {}
}

# ==========================================
# STATUS
# ==========================================
function Show-SAMStatus {
    $TaskList = @(
        $TASK_AGENT_SYNC,
        $TASK_C2_DAY,
        $TASK_C2_NIGHT,
        $TASK_FULL_SYNC,
        $TASK_UPDATE_CHECK,
        $TASK_LOG_ROTATE
    )

    Write-Host "`n╔══════════════════════════════════════╗" -ForegroundColor Cyan
    Write-Host "║   S.A.M. SCHEDULER STATUS v$SAM_VERSION   ║" -ForegroundColor Cyan
    Write-Host "╚══════════════════════════════════════╝`n" -ForegroundColor Cyan

    foreach ($TName in $TaskList) {
        $Task = Get-ScheduledTask -TaskName $TName -ErrorAction SilentlyContinue
        $Info = Get-ScheduledTaskInfo -TaskName $TName -ErrorAction SilentlyContinue
        if ($Task) {
            $State     = $Task.State
            $LastRun   = if ($Info.LastRunTime -and $Info.LastRunTime.Year -gt 2000) { $Info.LastRunTime.ToString("yyyy-MM-dd HH:mm") } else { "Never" }
            $NextRun   = if ($Info.NextRunTime -and $Info.NextRunTime.Year -gt 2000) { $Info.NextRunTime.ToString("yyyy-MM-dd HH:mm") } else { "N/A" }
            $LastResult = $Info.LastTaskResult
            $StatusColor = if ($State -eq "Ready" -or $State -eq "Running") { "Green" } else { "Red" }
            Write-Host "  [✓] $TName" -ForegroundColor $StatusColor
            Write-Host "      State: $State | Last: $LastRun | Next: $NextRun | Result: $LastResult" -ForegroundColor Gray
        } else {
            Write-Host "  [✗] $TName  (NOT INSTALLED)" -ForegroundColor Red
        }
    }
    Write-Host ""
}

# ==========================================
# UNINSTALL
# ==========================================
function Uninstall-SAM {
    Write-SAMLog "=== SAM Uninstall requested ==="
    foreach ($TName in @($TASK_AGENT_SYNC,$TASK_C2_DAY,$TASK_C2_NIGHT,$TASK_FULL_SYNC,$TASK_UPDATE_CHECK,$TASK_LOG_ROTATE)) {
        $Task = Get-ScheduledTask -TaskName $TName -ErrorAction SilentlyContinue
        if ($Task) {
            Unregister-ScheduledTask -TaskName $TName -Confirm:$false -ErrorAction SilentlyContinue | Out-Null
            Write-SAMLog "Removed task: $TName"
        }
    }
    Write-Host "SAM uninstall complete." -ForegroundColor Yellow
}
