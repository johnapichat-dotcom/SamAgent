# ============================================================
# LOGGING
# ============================================================
function Write-SAMLog {
    param(
        [string]$Message,
        [ValidateSet('INFO','WARN','ERROR','DEBUG')]
        [string]$Level = 'INFO'
    )
    $ts   = Get-Date -Format 'yyyy-MM-dd HH:mm:ss'
    $line = "$ts [$Level] [Update] $Message"
    try {
        Add-Content -Path $LogFile -Value $line -Encoding UTF8 -ErrorAction SilentlyContinue
    } catch {}
    if ($Level -in 'WARN','ERROR') {
        Write-Host $line -ForegroundColor $(if ($Level -eq 'ERROR') {'Red'} else {'Yellow'})
    } else {
        Write-Host $line
    }
}

# ============================================================
# DOWNLOAD HELPER
# ✅ v3.3.2: เพิ่ม Unblock-File หลัง Download ทุกครั้ง
#            แก้ปัญหา Zone.Identifier (Mark of the Web)
#            ที่ทำให้ไฟล์ที่โหลดจาก Internet รันไม่ได้
# ============================================================
function Get-SAMFile {
    param(
        [string]$Url,
        [string]$OutPath,
        [int]   $MinKB   = 5,
        [int]   $Retries = 3
    )
    for ($i = 1; $i -le $Retries; $i++) {
        try {
            Invoke-WebRequest -Uri $Url -OutFile $OutPath `
                -UseBasicParsing -TimeoutSec 60 -ErrorAction Stop

            # ✅ FIX v3.3.2: Unblock ไฟล์ทันทีหลัง Download
            # ป้องกัน "You do not have permission to open this file"
            # และ Execution Policy block จาก Zone.Identifier stream
            Unblock-File -Path $OutPath -ErrorAction SilentlyContinue

            if ((Test-Path $OutPath) -and
                (Get-Item $OutPath).Length -ge ($MinKB * 1KB)) {
                $sizeKB = [math]::Round((Get-Item $OutPath).Length / 1KB, 1)
                Write-SAMLog "Downloaded+Unblocked: $(Split-Path $OutPath -Leaf) (${sizeKB}KB)"
                return $true
            }
            Write-SAMLog "File too small (attempt $i): $OutPath" -Level 'WARN'
        } catch {
            Write-SAMLog "Download fail attempt $i/${Retries}: $Url — $($_.Exception.Message)" -Level 'WARN'
            if ($i -lt $Retries) { Start-Sleep -Seconds (10 * $i) }
        }
    }
    Write-SAMLog "FAILED to download: $Url" -Level 'ERROR'
    return $false
}

# ============================================================
# ✅ FIX PERMISSION — Reset SpbKey.txt ให้ Administrators อ่านได้
# ============================================================
function Repair-SpbKeyPermission {
    try {
        $result = & icacls $KeyFile /reset 2>&1
        Write-SAMLog "SpbKey.txt permission reset: $result"
        & icacls $KeyFile /grant "SYSTEM:(F)" 2>&1 | Out-Null
        & icacls $KeyFile /grant "Administrators:(F)" 2>&1 | Out-Null
        Write-SAMLog "SpbKey.txt permission set: SYSTEM + Administrators (F)"
        return $true
    } catch {
        Write-SAMLog "Permission repair failed: $($_.Exception.Message)" -Level 'WARN'
        return $false
    }
}

# ============================================================
# ✅ SpbKey.txt: ตรวจสอบและสร้างถ้าหาย
# ============================================================
function Confirm-SpbKey {
    param([bool]$ForceCreate = $false)

    $needCreate = $false

    if (-not (Test-Path $KeyFile)) {
        Write-SAMLog "SpbKey.txt not found" -Level 'WARN'
        $needCreate = $true
    } else {
        try {
            $keyRaw     = Get-Content $KeyFile -Raw -Encoding UTF8 -ErrorAction Stop
            $keyContent = if ($null -ne $keyRaw) { $keyRaw.Trim() } else { '' }

            if ([string]::IsNullOrWhiteSpace($keyContent)) {
                Write-SAMLog "SpbKey.txt is empty" -Level 'WARN'
                $needCreate = $true
            } elseif ($keyContent.Length -lt 100) {
                Write-SAMLog "SpbKey.txt content too short ($($keyContent.Length) chars)" -Level 'WARN'
                $needCreate = $true
            } elseif (-not ($keyContent.StartsWith('eyJ') -or $keyContent.StartsWith('sb_'))) {
                Write-SAMLog "SpbKey.txt format invalid (not JWT/sb_ format)" -Level 'WARN'
                $needCreate = $true
            } else {
                Write-SAMLog "SpbKey.txt OK ($($keyContent.Length) chars)"
                return $true
            }
        } catch {
            if ($_.Exception.Message -like '*denied*' -or
                $_.Exception.Message -like '*Access*') {
                Write-SAMLog "SpbKey.txt permission issue — repairing..." -Level 'WARN'
                Repair-SpbKeyPermission | Out-Null
                try {
                    $keyRaw     = Get-Content $KeyFile -Raw -Encoding UTF8 -ErrorAction Stop
                    $keyContent = if ($null -ne $keyRaw) { $keyRaw.Trim() } else { '' }
                    if (-not [string]::IsNullOrWhiteSpace($keyContent) -and
                        $keyContent.Length -ge 100 -and
                        ($keyContent.StartsWith('eyJ') -or $keyContent.StartsWith('sb_'))) {
                        Write-SAMLog "SpbKey.txt OK after permission repair ($($keyContent.Length) chars)"
                        return $true
                    }
                    $needCreate = $true
                } catch {
                    Write-SAMLog "SpbKey.txt still unreadable after repair — will recreate" -Level 'WARN'
                    $needCreate = $true
                }
            } else {
                Write-SAMLog "SpbKey.txt read error: $($_.Exception.Message)" -Level 'WARN'
                $needCreate = $true
            }
        }
    }

    if ($needCreate -or $ForceCreate) {
        try {
            if (Test-Path $KeyFile) {
                try {
                    Copy-Item $KeyFile "$KeyFile.bak" -Force -ErrorAction SilentlyContinue
                    Write-SAMLog "SpbKey.txt backed up → SpbKey.txt.bak"
                } catch {
                    & icacls $KeyFile /reset 2>&1 | Out-Null
                    & icacls $KeyFile /grant "Administrators:(F)" 2>&1 | Out-Null
                    Remove-Item $KeyFile -Force -ErrorAction SilentlyContinue
                    Write-SAMLog "SpbKey.txt removed (permission reset)"
                }
            }

            [IO.File]::WriteAllText($KeyFile, $EmbeddedKey, [Text.Encoding]::UTF8)

            try {
                & icacls $KeyFile /grant "SYSTEM:(F)" 2>&1 | Out-Null
                & icacls $KeyFile /grant "Administrators:(F)" 2>&1 | Out-Null
                Write-SAMLog "SpbKey.txt permission set OK"
            } catch {
                Write-SAMLog "Permission set skipped: $($_.Exception.Message)" -Level 'WARN'
            }

            $verifyRaw     = Get-Content $KeyFile -Raw -Encoding UTF8 -ErrorAction Stop
            $verifyContent = if ($null -ne $verifyRaw) { $verifyRaw.Trim() } else { '' }

            if ($verifyContent -eq $EmbeddedKey) {
                Write-SAMLog "SpbKey.txt created/restored OK ($($EmbeddedKey.Length) chars)"
                return $true
            } else {
                Write-SAMLog "SpbKey.txt content mismatch after write" -Level 'ERROR'
                return $false
            }
        } catch {
            Write-SAMLog "Failed to create SpbKey.txt: $($_.Exception.Message)" -Level 'ERROR'
            return $false
        }
    }
    return $false
}

# ============================================================
# SELF-HEALING
# ✅ v3.3.2: เพิ่ม Unblock-File หลัง Move-Item ทุกไฟล์ที่ Heal
# ============================================================
function Invoke-SelfHealing {
    Write-SAMLog "--- Self-Healing Check ---"
    $healed = 0

    # 1. SpbKey.txt
    Write-SAMLog "Checking SpbKey.txt..."
    $keyOK = Confirm-SpbKey
    if ($keyOK) { $healed++ }
    else { Write-SAMLog "SpbKey.txt healing FAILED — agent may not work" -Level 'ERROR' }

    # 2. SAM_Agent.ps1
    $agentFile = "$WorkDir\SAM_Agent.ps1"
    if (-not (Test-Path $agentFile) -or
        (Get-Item $agentFile -ErrorAction SilentlyContinue).Length -lt ($MIN_AGENT * 1KB)) {
        Write-SAMLog "Healing SAM_Agent.ps1..." -Level 'WARN'
        $tmp = "$WorkDir\SAM_Agent_heal.ps1"
        if (Get-SAMFile $URL_AGENT $tmp $MIN_AGENT) {
            Move-Item $tmp $agentFile -Force
            # ✅ v3.3.2: Unblock หลัง Move-Item (Zone.Identifier อาจติดมากับ tmp)
            Unblock-File -Path $agentFile -ErrorAction SilentlyContinue
            Write-SAMLog "SAM_Agent.ps1 healed+unblocked OK"
            $healed++
        }
    } else {
        Write-SAMLog "SAM_Agent.ps1 OK ($([math]::Round((Get-Item $agentFile).Length/1KB,1))KB)"
        $healed++
    }

    # 3. SAM_Scheduler.ps1
    $schedulerFile = "$WorkDir\SAM_Scheduler.ps1"
    if (-not (Test-Path $schedulerFile) -or
        (Get-Item $schedulerFile -ErrorAction SilentlyContinue).Length -lt ($MIN_SCHEDULER * 1KB)) {
        Write-SAMLog "Healing SAM_Scheduler.ps1..." -Level 'WARN'
        $tmp = "$WorkDir\SAM_Scheduler_heal.ps1"
        if (Get-SAMFile $URL_SCHEDULER $tmp $MIN_SCHEDULER) {
            Move-Item $tmp $schedulerFile -Force
            # ✅ v3.3.2: Unblock หลัง Move-Item
            Unblock-File -Path $schedulerFile -ErrorAction SilentlyContinue
            Write-SAMLog "SAM_Scheduler.ps1 healed+unblocked OK"
            $healed++
        }
    } else {
        Write-SAMLog "SAM_Scheduler.ps1 OK ($([math]::Round((Get-Item $schedulerFile).Length/1KB,1))KB)"
        $healed++
    }

    # 4. SystemConfig.json
    $configFile = "$WorkDir\SystemConfig.json"
    if (-not (Test-Path $configFile)) {
        Write-SAMLog "SystemConfig.json missing — please reinstall" -Level 'ERROR'
    } else {
        Write-SAMLog "SystemConfig.json OK"
        $healed++
    }

    Write-SAMLog "Self-Healing: $healed/4 files OK"
    return $healed
}

# ============================================================
# VERSION CHECK
# ============================================================
function Get-RemoteVersion {
    try {
        $tmp = "$env:TEMP\sam_version_check.txt"
        Invoke-WebRequest -Uri $URL_VERSION -OutFile $tmp `
            -UseBasicParsing -TimeoutSec 15 -ErrorAction Stop
        $ver = (Get-Content $tmp -Raw).Trim()
        Remove-Item $tmp -Force -ErrorAction SilentlyContinue
        return $ver
    } catch {
        Write-SAMLog "Cannot reach version server: $($_.Exception.Message)" -Level 'WARN'
        return $null
    }
}

function Get-LocalVersion {
    if (Test-Path $VersionFile) {
        return (Get-Content $VersionFile -Raw).Trim()
    }
    return '0.0.0'
}

function Compare-Version {
    param([string]$Local, [string]$Remote)
    try {
        return ([version]$Remote -gt [version]$Local)
    } catch { return $false }
}

# UPDATE FILES
# ✅ v3.3.2: เพิ่ม Unblock-File หลัง Move-Item ทุกไฟล์ที่ Replace
# ============================================================
function Invoke-UpdateFiles {
    param([string]$RemoteVersion)

    Write-SAMLog "Updating to v$RemoteVersion..."
    $updated = 0

    # Backup
    $backupDir = "$WorkDir\Backup_$(Get-Date -Format 'yyyyMMdd_HHmmss')"
    New-Item -ItemType Directory -Force -Path $backupDir | Out-Null
    foreach ($file in @('SAM_Agent.ps1','SAM_Scheduler.ps1','SAM_Update.ps1')) {
        $src = "$WorkDir\$file"
        if (Test-Path $src) {
            Copy-Item $src "$backupDir\$file" -Force -ErrorAction SilentlyContinue
        }
    }
    Write-SAMLog "Backup created: $backupDir"

    # Download (Get-SAMFile จะ Unblock อัตโนมัติแล้ว)
    $downloads = @(
        @{ Url=$URL_AGENT;     Out="$WorkDir\SAM_Agent_new.ps1";     MinKB=$MIN_AGENT },
        @{ Url=$URL_SCHEDULER; Out="$WorkDir\SAM_Scheduler_new.ps1"; MinKB=$MIN_SCHEDULER },
        @{ Url=$URL_UPDATE;    Out="$WorkDir\SAM_Update_new.ps1";    MinKB=$MIN_UPDATE }
    )

    $allOK = $true
    foreach ($dl in $downloads) {
        if (-not (Get-SAMFile $dl.Url $dl.Out $dl.MinKB)) {
            $allOK = $false
            Write-SAMLog "Download failed: $($dl.Out)" -Level 'ERROR'
        }
    }

    if (-not $allOK) {
        Write-SAMLog "Update aborted — download error" -Level 'ERROR'
        return $false
    }

    # Replace
    $replacePairs = @(
        @{ New="$WorkDir\SAM_Agent_new.ps1";     Final="$WorkDir\SAM_Agent.ps1" },
        @{ New="$WorkDir\SAM_Scheduler_new.ps1"; Final="$WorkDir\SAM_Scheduler.ps1" },
        @{ New="$WorkDir\SAM_Update_new.ps1";    Final="$WorkDir\SAM_Update.ps1" }
    )

    foreach ($pair in $replacePairs) {
        try {
            Move-Item $pair.New $pair.Final -Force
            # ✅ v3.3.2: Unblock อีกรอบหลัง Move-Item (double-safe)
            Unblock-File -Path $pair.Final -ErrorAction SilentlyContinue
            Write-SAMLog "Replaced+Unblocked: $(Split-Path $pair.Final -Leaf)"
            $updated++
        } catch {
            Write-SAMLog "Replace failed: $($pair.Final) — $($_.Exception.Message)" -Level 'ERROR'
        }
    }

    # Update version
    [IO.File]::WriteAllText($VersionFile, $RemoteVersion, [Text.Encoding]::UTF8)
    Write-SAMLog "Version updated: $RemoteVersion"

    # ============================================================
    # POST-UPDATE: Run 3 ไฟล์หลัง Update สำเร็จ
    # ============================================================
    Write-SAMLog "--- Post-update: Running 3 scripts ---"

    # [1/3] SAM_Scheduler.ps1 -Setup
    Write-SAMLog "Post-update [1/3]: SAM_Scheduler.ps1 -Setup"
    try {
        $schedulerPath = "$WorkDir\SAM_Scheduler.ps1"
        if (Test-Path $schedulerPath) {
            & powershell -ExecutionPolicy Bypass `
                -File $schedulerPath -Setup
            Write-SAMLog "Scheduler Setup OK"

            $bats = @('SAM_ForceSync.bat','SAM_DeepScan.bat',
                      'SAM_Status.bat','SAM_Uninstall.bat',
                      'SAM_UpdateCheck.bat','SAM_C2Poll.bat')
            $missingBats = $bats | Where-Object { -not (Test-Path "$WorkDir\$_") }
            if ($missingBats) {
                Write-SAMLog "Missing .bat: $($missingBats -join ', ')" -Level 'WARN'
            } else {
                Write-SAMLog "All .bat files OK ($($bats.Count) files)"
            }
        } else {
            Write-SAMLog "Scheduler not found: $schedulerPath" -Level 'WARN'
        }
    } catch {
        Write-SAMLog "Scheduler Setup error: $($_.Exception.Message)" -Level 'WARN'
    }

    # [2/3] SpbKey.txt verify
    Write-SAMLog "Post-update [2/3]: SpbKey.txt verify"
    Confirm-SpbKey | Out-Null

    # Cleanup backups เก่า
    $backups = Get-ChildItem "$WorkDir\Backup_*" -Directory |
               Sort-Object Name -Descending | Select-Object -Skip 3
    foreach ($b in $backups) {
        Remove-Item $b.FullName -Recurse -Force -ErrorAction SilentlyContinue
        Write-SAMLog "Removed old backup: $($b.Name)"
    }

    Write-SAMLog "Update complete: $updated/3 files replaced"
    return ($updated -eq 3)
}

# ============================================================
# MAIN
# ============================================================
function Main {
    Write-SAMLog "=== SAM Update Manager v$SAM_UPDATE_VERSION ==="

    if (-not (Test-Path $WorkDir)) {
        Write-SAMLog "WorkDir not found: $WorkDir — abort" -Level 'ERROR'
        exit 1
    }

    # STEP 1: Self-Healing
    Write-SAMLog "--- STEP 1: Self-Healing ---"
    Invoke-SelfHealing | Out-Null

    # STEP 2: Critical SpbKey Check
    Write-SAMLog "--- STEP 2: Critical File Check ---"
    $keyOK = $false
    if (Test-Path $KeyFile) {
        try {
            $keyRaw     = Get-Content $KeyFile -Raw -Encoding UTF8 -ErrorAction Stop
            $keyContent = if ($null -ne $keyRaw) { $keyRaw.Trim() } else { '' }

            if (-not [string]::IsNullOrWhiteSpace($keyContent) -and
                $keyContent.Length -ge 100 -and
                ($keyContent.StartsWith('eyJ') -or $keyContent.StartsWith('sb_'))) {
                Write-SAMLog "SpbKey.txt VERIFIED OK ($($keyContent.Length) chars)"
                $keyOK = $true
            } else {
                Write-SAMLog "SpbKey.txt invalid — recreating" -Level 'WARN'
                $keyOK = Confirm-SpbKey -ForceCreate $true
            }
        } catch {
            if ($_.Exception.Message -like '*denied*' -or
                $_.Exception.Message -like '*Access*') {
                Write-SAMLog "SpbKey.txt permission issue — repairing" -Level 'WARN'
                Repair-SpbKeyPermission | Out-Null
                $keyOK = Confirm-SpbKey -ForceCreate $true
            } else {
                $keyOK = Confirm-SpbKey -ForceCreate $true
            }
        }
    } else {
        $keyOK = Confirm-SpbKey -ForceCreate $true
    }

    if (-not $keyOK) {
        Write-SAMLog "SpbKey.txt CRITICAL FAIL — agent may not sync" -Level 'ERROR'
    }

    # STEP 3: Version Check
    Write-SAMLog "--- STEP 3: Version Check ---"
    $localVer  = Get-LocalVersion
    $remoteVer = Get-RemoteVersion

    if ($null -eq $remoteVer) {
        Write-SAMLog "Cannot check remote version — skip update" -Level 'WARN'
        Write-SAMLog "=== Update Manager Done (no network) ==="
        exit 0
    }

    Write-SAMLog "Local:  v$localVer"
    Write-SAMLog "Remote: v$remoteVer"

    # STEP 4: Update
    if (Compare-Version $localVer $remoteVer) {
        Write-SAMLog "--- STEP 4: Update Available → v$remoteVer ---"
        $updateOK = Invoke-UpdateFiles -RemoteVersion $remoteVer

        if ($updateOK) {
            Write-SAMLog "Update SUCCESS: v$localVer → v$remoteVer"

            # [3/3] SAM_Agent.ps1 -DeltaOnly
            Write-SAMLog "Post-update [3/3]: SAM_Agent.ps1 -DeltaOnly"
            try {
                $agentPath = "$WorkDir\SAM_Agent.ps1"
                if (Test-Path $agentPath) {
                    Start-Sleep -Seconds 3
                    & powershell -ExecutionPolicy Bypass `
                        -File $agentPath `
                        -Silent -DeltaOnly
                    Write-SAMLog "Agent DeltaSync OK — version confirmed in Supabase"
                } else {
                    Write-SAMLog "Agent not found: $agentPath" -Level 'WARN'
                }
            } catch {
                Write-SAMLog "Agent DeltaSync error: $($_.Exception.Message)" -Level 'WARN'
            }

            Write-SAMLog "=== Post-update complete: Scheduler + SpbKey + Agent ==="

        } else {
            Write-SAMLog "Update FAILED" -Level 'ERROR'
            exit 1
        }
    } else {
        Write-SAMLog "Already up-to-date (v$localVer)"
    }

    # STEP 5: Final Summary
    Write-SAMLog "--- STEP 5: Final Summary ---"
    $allOK = $true

    $files = @(
        @{ Name='SAM_Agent.ps1';    MinKB=$MIN_AGENT },
        @{ Name='SAM_Scheduler.ps1';MinKB=$MIN_SCHEDULER },
        @{ Name='SAM_Update.ps1';   MinKB=$MIN_UPDATE },
        @{ Name='SpbKey.txt';       MinKB=0 },
        @{ Name='SystemConfig.json';MinKB=0 }
    )

    foreach ($f in $files) {
        $path = "$WorkDir\$($f.Name)"
        if (Test-Path $path) {
            $sizeKB = [math]::Round((Get-Item $path).Length / 1KB, 2)
            if ($f.MinKB -gt 0 -and $sizeKB -lt $f.MinKB) {
                Write-SAMLog "⚠ $($f.Name) too small: ${sizeKB}KB" -Level 'WARN'
                $allOK = $false
            } else {
                Write-SAMLog "OK: $($f.Name) (${sizeKB}KB)"
            }
        } else {
            Write-SAMLog "MISSING: $($f.Name)" -Level 'ERROR'
            $allOK = $false
        }
    }

    if (Test-Path $KeyFile) {
        try {
            $keyRaw = Get-Content $KeyFile -Raw -Encoding UTF8 -ErrorAction Stop
            $keyLen = if ($null -ne $keyRaw) { $keyRaw.Trim().Length } else { 0 }
            if ($keyLen -ge 100) {
                Write-SAMLog "SpbKey.txt content OK ($keyLen chars)"
            } else {
                Write-SAMLog "SpbKey.txt content INVALID ($keyLen chars)" -Level 'ERROR'
                $allOK = $false
            }
        } catch {
            Write-SAMLog "SpbKey.txt read error in summary: $($_.Exception.Message)" -Level 'ERROR'
            $allOK = $false
        }
    }

    if ($allOK) {
        Write-SAMLog "=== Update Manager COMPLETED ==="
    } else {
        Write-SAMLog "=== Update Manager COMPLETED WITH WARNINGS ===" -Level 'WARN'
    }

    exit 0
}

# ============================================================
# ENTRY POINT
# ============================================================
try {
    Main
} catch {
    Write-SAMLog "Unhandled error: $($_.Exception.Message)" -Level 'ERROR'
    exit 1
